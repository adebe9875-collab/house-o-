function updateList() {

  const groups = document.querySelectorAll('.os-group');

    let visibleItemsInGroup = 0;

    items.forEach(item => {
      const itemType = item.getAttribute('data-type');
      const itemName = item.getAttribute('data-name');

      const matchesSearch = itemName.includes(query);

        item.style.display = 'flex';

      } else {
        item.style.display = 'none';
      }

  });
}

function speakArabic(text) {
  if (!window.speechSynthesis) return;
  window.speechSynthesis.cancel();

  const cleaned = text
    .replace(/[\u{1F300}-\u{1FAFF}\u{2600}-\u{27BF}\u{200D}\uFE0F]/gu, '')
    .replace(/\n+/g, '. ')
    .trim();

  // نقسّم النص لجمل قصيرة، لأن بعض المتصفحات تقطع النطق إذا كان النص طويلاً جداً في مرة واحدة
  const chunks = cleaned.split(/(?<=[.!؟?])\s+/).filter(c => c.trim().length > 0);

  chunks.forEach(chunk => {
    const utterance = new SpeechSynthesisUtterance(chunk);
    utterance.lang = 'ar-SA';
    if (arabicVoice) utterance.voice = arabicVoice;
    utterance.rate = 1;
    window.speechSynthesis.speak(utterance);
  });
}

async function computeSHA256(text) {
  const encoder = new TextEncoder();
  const data = encoder.encode(text);
  const hashBuffer = await crypto.subtle.digest('SHA-256', data);
  const hashArray = Array.from(new Uint8Array(hashBuffer));
  return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
}
